{# Simple, dependency-free charts. Single-series bars: one hue, value at the bar tip, sample size beside every label, hover shows the exact value. The tables on the page carry the same numbers, so nothing depends on the chart alone. #} {% macro fmtv(v, fmt) -%} {%- if v is none -%}— {%- elif fmt == 'pct' -%}{{ v|pct }} {%- elif fmt == 'x' -%}{{ '%.2f'|format(v) }}× {%- else -%}{{ v|num(2) }}{%- endif -%} {%- endmacro %} {% macro hbar(rows, value_key, title, fmt='num', label_key='label', n_key='n', note='') %} {% set vals = rows|map(attribute=value_key)|reject('none')|list %} {% set maxv = (vals|max) if vals else 0 %} {% endmacro %} {% macro line(points, title, value_label) %} {# points: list of (label, value) with value not none; needs >= 2 #} {% set W, H, P = 560, 160, 28 %} {% set vals = points|map(attribute=1)|list %} {% set maxv = vals|max %}{% set minv = 0 %} {% set span = (maxv - minv) or 1 %} {% set step = (W - 2 * P) / ((points|length - 1) or 1) %} {% endmacro %}
